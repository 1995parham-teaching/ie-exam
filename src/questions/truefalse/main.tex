\قسمت{صحیح/غلط}

درستی یا نادرستی هر یک از عبارت‌های زیر را مشخص کنید.
برای مواردی که نادرست می‌دانید، در یک جمله دلیلش را هم بنویسید؛ پاسخ بدون دلیل نمره ندارد.

\شروع{شمارش}

\فقره قطعه کدی که در ادامه آمده است، \متن‌سیاه{تمام} المان‌هایی که کلاس \متن‌لاتین{example} دارند، را برمی‌گرداند.

\begin{latin}
\begin{minted}{js}
document.querySelector(".example")
\end{minted}
\end{latin}

\فقره برای صحت‌سنجی مستندات \متن‌لاتین{JSON} می‌توان ساختار و قوانین صحت‌سنجی را در قالب مستنداتی با ساختار \متن‌لاتین{JSON Validator} توصیف نمود.

\فقره داکر بر پایه‌ی ویژگی‌های \متن‌سیاه{منحصر به فرد} هسته لینوکس شکل گرفته است.

\فقره \متن‌لاتین{margin}های عمودی المان‌ها به صورت پیش‌فرض در \متن‌لاتین{CSS} جمع نمی‌شوند.

\پایان{شمارش}


\نمره{۱}

% پاسخ پیشنهادی است و پیش از استفاده در تصحیح نیاز به بازبینی دارد.
\begin{پاسخ}

\شروع{شمارش}
\فقره \متن‌سیاه{نادرست.} \متن‌لاتین{querySelector} تنها نخستین المان منطبق را برمی‌گرداند؛ برای همه‌ی آن‌ها \متن‌لاتین{querySelectorAll} لازم است.
\فقره \متن‌سیاه{نادرست.} نام آن ساختار \متن‌لاتین{JSON Schema} است، نه \متن‌لاتین{JSON Validator}؛ \متن‌لاتین{validator} ابزاری است که سند را با \متن‌لاتین{schema} می‌سنجد.
\فقره این گزینه دو خوانش دارد: قابلیت‌های مورد استفاده‌ی داکر (\متن‌لاتین{namespace} و \متن‌لاتین{cgroup}) مختص لینوکس هستند، اما ایده‌ی جداسازی منحصر به لینوکس نیست (\متن‌لاتین{Jail} در \متن‌لاتین{FreeBSD} و \متن‌لاتین{Zones} در \متن‌لاتین{Solaris}). پاسخ همراه با استدلال درست، نمره‌ی کامل دارد.
\فقره \متن‌سیاه{درست.} حاشیه‌های عمودی مجاور در هم فرو می‌روند\پانویس{Margin Collapsing} و بزرگ‌ترین آن‌ها اعمال می‌شود، پس جمع نمی‌شوند.
\پایان{شمارش}

هر مورد یک چهارم نمره را دارد.

\end{پاسخ}
