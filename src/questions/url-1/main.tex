\قسمت{شکستن \lr{URL}}

با در نظر گرفتن \متن‌لاتین{URL} زیر، قسمت‌های مختلف آن را مطابق با آنچه در درس آمده است توضیح دهید:

\begin{latin}
\begin{minted}[]{output}
  postgresql://parham:secret@127.0.0.1:5432/
    hellodb?connect_timeout=10&application_name=morpheus
\end{minted}
\end{latin}

\نمره{۱}

% پاسخ پیشنهادی است و پیش از استفاده در تصحیح نیاز به بازبینی دارد.
\begin{پاسخ}

\شروع{فقرات}
\فقره \متن‌لاتین{postgresql}: طرح\پانویس{Scheme} یا پروتکل
\فقره \متن‌لاتین{parham}: نام کاربری
\فقره \متن‌لاتین{secret}: رمز عبور
\فقره \متن‌لاتین{127.0.0.1}: میزبان
\فقره \متن‌لاتین{5432}: پورت
\فقره \متن‌لاتین{/hellodb}: مسیر، که اینجا نام پایگاه داده است
\فقره \متن‌لاتین{connect\_timeout=10} و \متن‌لاتین{application\_name=morpheus}: پرس‌وجو\پانویس{Query}
\پایان{فقرات}

\end{پاسخ}
